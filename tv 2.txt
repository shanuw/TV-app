        function startPlay(url, name) {
            document.getElementById('player-container').style.display = 'block';

            document.getElementById('external-link').href = url;

            // සයිට් එකක් නම් (PHP ලින්ක් නම්) කෙලින්ම ප්ලේ වෙන්නේ නැහැ, ඒ නිසා අපි චෙක් කරනවා
            if(url.includes('.m3u8')) {
                player.src({ src: url, type: 'application/x-mpegURL' });
                player.play();
            } else {
                // PHP හෝ වෙනත් ලින්ක් එකක් නම්, යූසර්ට Direct බට්න් එක ඔබන්න කියනවා
                player.pause();
                alert("මෙම නාලිකාව 'Data Free' බැවින් ආරක්ෂිත හේතු මත එය වෙනම ටැබ් එකක විවෘත වේ.");
                window.open(url, '_blank');
            }
            window.scrollTo({top: 0, behavior: 'smooth'});
        }
