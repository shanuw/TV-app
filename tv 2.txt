        var myPlayer = videojs('tv-player');

        function changeTV(url, name, isDirect) {
            document.getElementById('ch-name').innerText = name;
            
            if(isDirect) {
                // PHP හෝ වෙනත් security ලින්ක් නම් බට්න් එක පෙන්වන්න
                myPlayer.pause();
                document.getElementById('direct-btn-area').style.display = 'block';
                document.getElementById('external-link').href = url;
                window.open(url, '_blank'); // අලුත් ටැබ් එකකත් open කරයි
            } else {
                // සාමාන්‍ය M3U8 ලින්ක් නම් player එකේ ප්ලේ කරන්න
                document.getElementById('direct-btn-area').style.display = 'none';
                myPlayer.src({ src: url, type: 'application/x-mpegURL' });
                myPlayer.play();
            }
            window.scrollTo({top: 0, behavior: 'smooth'});
        }
